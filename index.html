function getGPS(){ const msg = document.getElementById('orderMsg') || document.getElementById('gpsText'); const state = document.getElementById('gpsState'); if (!navigator.geolocation) { if (msg) msg.textContent = 'Бу қурилма GPSни қўлламайди.'; if (state) state.textContent = 'Йўқ'; return; } if (msg) msg.textContent = '📍 GPS аниқланмоқда...'; navigator.geolocation.getCurrentPosition( pos => { gps = { lat: pos.coords.latitude, lon: pos.coords.longitude }; if (state) state.textContent = 'OK'; if (msg) { msg.textContent = `📍 Манзил GPS аниқланди: ${gps.lat.toFixed(6)}, ${gps.lon.toFixed(6)}. Энди буюртмани юборинг.`; } }, err => { if (state) state.textContent = 'Йўқ'; if (msg) { msg.textContent = err.code === 1 ? 'GPS рухсати берилмади. Браузерда Location → Allow қилинг.' : 'GPS аниқланмади. Қайта уриниб кўринг.'; } }, { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 } ); }
